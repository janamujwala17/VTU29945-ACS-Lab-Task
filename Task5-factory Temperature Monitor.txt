import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int N = sc.nextInt();
        int[] temp = new int[N];
        int[] result = new int[N];

        for (int i = 0; i < N; i++) {
            temp[i] = sc.nextInt();
            result[i] = -1;
        }

        Stack<Integer> stack = new Stack<>();
        for (int i = N - 1; i >= 0; i--) {

                       while (!stack.isEmpty() && stack.peek() <= temp[i]) {
                stack.pop();
            }


            if (!stack.isEmpty()) {
                result[i] = stack.peek();
            }

                       stack.push(temp[i]);
        }

        for (int i = 0; i < N; i++) {
            System.out.print(result[i]);
            if (i < N - 1) {
                System.out.print(" ");
            }
        }
    }
}
