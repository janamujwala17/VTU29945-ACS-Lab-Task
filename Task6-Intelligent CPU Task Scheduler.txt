import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int N = sc.nextInt();
        int K = sc.nextInt();

        int[] freq = new int[26];

        for (int i = 0; i < N; i++) {
            char task = sc.next().charAt(0);
            freq[task - 'A']++;
        }

                int maxFreq = 0;
        for (int f : freq) {
            maxFreq = Math.max(maxFreq, f);
        }
        int maxCount = 0;
        for (int f : freq) {
            if (f == maxFreq) {
                maxCount++;
            }
        }

                int result = (maxFreq - 1) * (K + 1) + maxCount;

        result = Math.max(result, N);

        System.out.println(result);
    }
}
